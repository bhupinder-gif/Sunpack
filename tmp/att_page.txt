function xlsxLib(){ return XLSX_P = XLSX_P || new Promise((res, rej) => { if (window.XLSX) return res(window.XLSX); const s = document.createElement('script'); s.src = 'https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js'; s.onload = () => res(window.XLSX); s.onerror = () => { XLSX_P = null; rej(new Error('Could not load the Excel reader — check internet and try again')); }; document.head.appendChild(s); }); }
const tm = v => { if (v == null || v === '') return ''; if (typeof v === 'number') { const mins = Math.round((v % 1) * 1440); return pad(Math.floor(mins/60))+':'+pad(mins%60); } const m = String(v).trim().match(/^(\d{1,2})[:.](\d{2})/); return m ? pad(+m[1])+':'+m[2] : ''; };
function parseTN(rows){
  let date = '', time = '', hi = -1;
  rows.forEach((r, i) => { const line = r.map(x => String(x ?? '')).join(' ');
    const m = line.match(/Date\s*:\s*(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})/i); if (m && !date) date = m[3]+'-'+pad(+m[2])+'-'+pad(+m[1]);
    const t = line.match(/Time\s*:\s*\S+\s+(\d{1,2}:\d{2})/i); if (t && !time) time = t[1];
    if (hi < 0 && r.some(x => /^pay\s*code$/i.test(String(x ?? '').trim()))) hi = i; });
  if (!date) { const t = rows.map(r=>r.join(' ')).join(' ').match(/Report on\s+(\d{1,2})-(\w{3})-(\d{4})/i); if (t) { const mi = ['jan','feb','mar','apr','may','jun','jul','aug','sep','oct','nov','dec'].indexOf(t[2].toLowerCase()); if (mi > -1) date = t[3]+'-'+pad(mi+1)+'-'+pad(+t[1]); } }
  if (hi < 0) throw new Error('This is not the Time Notifier Daily Attendance Report (no "Pay Code" column)');
  const H = rows[hi].map(x => String(x ?? '').trim().toLowerCase()), ix = k => H.findIndex(h => h === k);
  const iC = ix('pay code'), iN = ix('name'), iI = ix('in time'), iO = ix('out time');
  if (iI < 0) throw new Error('No "In Time" column in the file');
  const out = [];
  rows.slice(hi+1).forEach(r => { const code = String(r[iC] ?? '').trim(); if (!/^\d+$/.test(code)) return;
    out.push({ code, name: String(r[iN] ?? '').trim(), in: tm(r[iI]), out: iO > -1 ? tm(r[iO]) : '' }); });
  return { date, time, rows: out };
}

/* ---------- manpower plan ---------- */
function renderPlan(){
  const ed = S.access.sir, P = S.plan || [], tot = P.reduce((t,p)=>t+(+p.req||0),0);
  const links = ['Machine','Any TF','Always'];
  let h = `<div style="overflow:auto"><table class="tbl" id="planT"><thead><tr><th>Area</th><th>Role</th><th>Machine</th><th class="r">Required</th><th>Plan link</th><th>Note</th>${ed?'<th></th>':''}</tr></thead><tbody>`;
  h += P.map((p,i) => ed ? `<tr><td><select data-f="area">${S.areas.map(a=>`<option ${a===p.area?'selected':''}>${esc(a)}</option>`).join('')}</select></td><td><input data-f="role" value="${esc(p.role)}"></td><td><input data-f="machine" value="${esc(p.machine)}"></td><td><input data-f="req" type="number" min="0" value="${p.req}" style="width:80px"></td><td><select data-f="link">${links.map(x=>`<option ${x===p.link?'selected':''}>${x}</option>`).join('')}</select></td><td><input data-f="note" value="${esc(p.note)}"></td><td><button class="btn red sm" data-del="${i}"><i data-lucide="x"></i></button></td></tr>`
    : `<tr><td>${esc(p.area)}</td><td>${esc(p.role)}</td><td>${esc(p.machine)}</td><td class="r"><b>${p.req}</b></td><td>${esc(p.link)}</td><td>${esc(p.note)}</td></tr>`).join('');
  h += `</tbody><tfoot><tr><td colspan="3"><b>Total per shift (all machines running)</b></td><td class="r"><b>${tot}</b></td><td colspan="${ed?3:2}"></td></tr></tfoot></table></div>`;
  if (ed) h += `<div style="display:flex;gap:8px;margin-top:12px"><button class="btn ghost" id="plAdd"><i data-lucide="plus"></i>Add row</button><button class="btn primary" id="plSave"><i data-lucide="save"></i>Save plan</button></div>`;
  else h += '<div class="sub" style="margin-top:10px">Only Bhupinder Sir and Rishu Sir change these numbers.</div>';
  $('planBox').innerHTML = h; icons();
  if (!ed) return;
  const read = () => [...$('planT').querySelectorAll('tbody tr')].map(tr => { const o = {}; tr.querySelectorAll('[data-f]').forEach(x => o[x.dataset.f] = x.value); return o; });
  $('planBox').querySelectorAll('[data-del]').forEach(b => b.onclick = () => { S.plan = read(); S.plan.splice(+b.dataset.del, 1); renderPlan(); });
  $('plAdd').onclick = () => { S.plan = read(); S.plan.push({ area: S.areas[0], role: 'Labour', machine: '', req: 1, link: 'Always', note: '' }); renderPlan(); };
  $('plSave').onclick = async ev => { const btn = ev.currentTarget; btn.disabled = true;
    try { const r = await call('attPlanSave', S.token, read()); S.plan = r.plan; toast('Plan saved'); renderPlan(); } catch (e) { fail(e); } finally { btn.disabled = false; } };
}
