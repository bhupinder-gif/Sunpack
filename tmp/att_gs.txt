/**
 * ATTENDANCE APP — separate app, opened from the portal at <web app URL>?app=att  (approved structure 26-Oct-2026)
 * Own workbook "Sunpack Attendance" (Script Property att_book), kept in the admin's own Drive — NOT in a shared folder,
 * because the Advance tab is only for Bhupinder Sir and Rishu Sir.
 *
 * Tabs: Manpower Plan · Labour · Daily · Extra Labour · Advance · Import Log   (banner row 1, title row 2, headers row 3, data from row 4)
 * Shifts: Day 8 AM–8 PM, Night 8 PM–8 AM.  Pay Code is the key (two "Sonu"s) — never match by name.
 * Required per shift (18): TF labour 5 (1 per machine) · TF operators 2 · sheet-line labour 4 · sheet-line operator 1 · grinders 4 · cleaning & loading 2.
 * Link to the production plan (PMS → Plan, released lines): a machine not planned for that shift needs 0 people;
 *   "Any TF" rows are needed when at least one thermoformer is planned; "Always" rows are always needed.
 *   If no plan is released for that date at all, the full need is shown with a note.
 * Time Notifier Excel: one-click import. In time with no Out = present, flagged orange ("No Out") for Shivam to fix.
 *   Shivam's manual entries (Source = Manual) are never overwritten by an import.
 * Access: Bhupinder Sir, Rishu Sir = everything (incl. advance list + balances)
 *         Shivam (PC) = attendance, labour list, extra labour, enter advances (cannot see them)
 *         Seema Mam  = enter advances only.   Everyone else (Sachin, Sonu …) = no access.
 */

const ATT_TABS = {
  plan: ['Manpower Plan', 'MANPOWER PLAN  —  people needed per shift · Plan Link: Machine = only when that machine is planned · Any TF = when any thermoformer is planned · Always',
    ['Area', 'Role', 'Machine', 'Required', 'Plan Link', 'Note']],
  lab: ['Labour', 'LABOUR  —  one row per person · Pay Code is the key (same as the Time Notifier machine)',
    ['Pay Code', 'Name', 'Area', 'Role', 'Default Shift', 'Phone', 'Active', 'Note', 'Added By', 'Added At']],
  day: ['Daily', 'DAILY  —  one row per person per shift · Source Manual = entered by Shivam, never overwritten by an import',
    ['Key', 'Date', 'Shift', 'Pay Code', 'Name', 'Area', 'Status', 'In', 'Out', 'Hours', 'Flag', 'Source', 'Note', 'Updated By', 'Updated At']],
  ext: ['Extra Labour', 'EXTRA LABOUR  —  extra people hired for a shift (not on the Labour list)',
    ['ID', 'Date', 'Shift', 'Area', 'Count', 'Names', 'Contractor', 'Note', 'Entered By', 'Entered At']],
  adv: ['Advance', 'ADVANCE  —  money given to labour and recovered · only Bhupinder Sir and Rishu Sir see this list',
    ['ID', 'Date', 'Pay Code', 'Name', 'Type', 'Amount', 'Note', 'Entered By', 'Entered At']],
  log: ['Import Log', 'IMPORT LOG  —  every Time Notifier Excel imported',
    ['At', 'By', 'File', 'Report Date', 'Rows', 'Present', 'No Out', 'New Labour', 'Kept Manual', 'Note']]
};
const ATT_ORDER = ['plan', 'lab', 'day', 'ext', 'adv', 'log'];
const ATT_AREAS = ['TF Labour', 'TF Operator', 'Sheet Line Labour', 'Sheet Line Operator', 'Grinder', 'Cleaning & Loading', 'Staff', 'Other'];
const ATT_STATUS = ['P', 'HD', 'A', 'L', 'WO'];   // present · half day · absent · leave · week off

/* ---------- workbook ---------- */
function attBook_(create) {
  const id = modProp_('att_book');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) {} }
  if (!create) throw new Error('NO_ACCESS: The Attendance workbook is not created yet — ask the admin to run setupAttendance()');
  const ss = SpreadsheetApp.create('Sunpack Attendance');   // stays in the admin's own Drive (not shared — advances are private)
  modProp_('att_book', ss.getId()); return ss;
}
function attSheet_(k) {
  const sh = attBook_(false).getSheetByName(ATT_TABS[k][0]);
  if (!sh) throw new Error('Attendance tab "' + ATT_TABS[k][0] + '" missing — run setupAttendance()');
  return sh;
}
function attH_(k) { return ATT_TABS[k][2]; }
function attRows_(k) {
  const sh = attSheet_(k), H = attH_(k), n = Math.min(H.length, sh.getMaxColumns());
  return modRows_(sh, n).map((r, i) => { const o = { _row: i + 4 }; H.forEach((h, j) => o[h] = j < n ? r[j] : ''); return o; }).filter(o => String(o[H[0]]).trim());
}
function attRowArr_(k, o) { return attH_(k).map(h => o[h] === undefined ? '' : o[h]); }
function attLock_(fn) { const l = LockService.getScriptLock(); l.waitLock(20000); try { return fn(); } finally { l.releaseLock(); } }
function attCode_(c) { return String(c == null ? '' : c).trim().replace(/^'/, '').replace(/^0+(?=\d)/, ''); }   // '0017' = '17' (key) — the shown code keeps its zeros
function attYmd_(d) { return d instanceof Date ? modFmt_(d, 'yyyy-MM-dd') : String(d || '').slice(0, 10); }
function attDate_(ymd) { const d = modYmd_(ymd); if (!d) throw new Error('Pick a date'); return d; }
function attShift_(s) { s = String(s || ''); if (!/^(Day|Night)$/.test(s)) throw new Error('Pick Day or Night shift'); return s; }
function attTime_(t) {
  if (t instanceof Date) return modFmt_(t, 'HH:mm');
  const m = String(t || '').trim().match(/^(\d{1,2})[:.](\d{2})/); return m ? m[1].padStart(2, '0') + ':' + m[2] : '';
}
function attHours_(a, b) {
  const p = s => { const m = String(s || '').match(/^(\d{2}):(\d{2})$/); return m ? +m[1] * 60 + +m[2] : null; };
  const x = p(a), y = p(b); if (x == null || y == null) return '';
  let d = y - x; if (d < 0) d += 1440; return Math.round(d / 6) / 10;
}
/* shift running now: 8 AM–8 PM = Day of today; 8 PM–midnight = Night of today; midnight–8 AM = Night of yesterday */
function attNow_() {
  const now = new Date(), h = +modFmt_(now, 'H'), d = modDay_(now);
  if (h < 8) return { ymd: modFmt_(new Date(d.getTime() - 864e5), 'yyyy-MM-dd'), shift: 'Night' };
  return { ymd: modFmt_(d, 'yyyy-MM-dd'), shift: h < 20 ? 'Day' : 'Night' };
}
/* shift from an In punch: 05:00–16:59 Day (same date) · 17:00–23:59 Night (same date) · 00:00–04:59 Night of the day before */
function attShiftOf_(ymd, inT) {
  const m = String(inT || '').match(/^(\d{2}):/); if (!m) return null; const h = +m[1];
  if (h >= 5 && h < 17) return { ymd, shift: 'Day' };
  if (h >= 17) return { ymd, shift: 'Night' };
  const d = modYmd_(ymd); return { ymd: modFmt_(new Date(d.getTime() - 864e5), 'yyyy-MM-dd'), shift: 'Night' };
}

/* ---------- who can do what ---------- */
function attWho_(u) {
  if (!u) return { sir: false, pc: false, adv: false };
  const s = settingsMap_(), me = normName_(u.name) + ' ' + normName_(u.username);
  const sirs = String(s['Attendance Full Access'] || 'bhupinder, rishu').split(/[,;]+/).map(normName_).filter(String);
  const sir = pcSir_(u) && sirs.some(x => me.indexOf(x) > -1);
  const pc = u.rtype === 'pc';
  const advL = String(s['Attendance Advance Users'] || 'seema').split(/[,;]+/).map(normName_).filter(String);
  const adv = sir || pc || u.rtype === 'ea' || advL.some(x => me.indexOf(x) > -1);
  return { sir, pc, att: sir || pc, adv };
}
function attUser_(token, need) {
  const u = getUser_(token); if (!u) throw new Error('SESSION_EXPIRED');
  const w = attWho_(u);
  if (!w.adv) throw new Error('NO_ACCESS: Attendance is run by Shivam (PC). Ask Bhupinder Sir if you need access.');
  if (need === 'att' && !w.att) throw new Error('NO_ACCESS: Only Shivam and the Sirs can open attendance.');
  if (need === 'sir' && !w.sir) throw new Error('NO_ACCESS: Only Bhupinder Sir and Rishu Sir can do this.');
  return { u, w };
}

/* ---------- labour list ---------- */
function attLabour_() {
  return attRows_('lab').map(r => ({ code: String(r['Pay Code']).trim(), key: attCode_(r['Pay Code']), name: String(r['Name']).trim(), area: String(r['Area'] || ''),
    role: String(r['Role'] || ''), shift: String(r['Default Shift'] || ''), phone: String(r['Phone'] || ''), active: !/^n/i.test(String(r['Active'] || 'Yes')), note: String(r['Note'] || ''), row: r._row }));
}

/* ---------- production plan → which machines run in a shift ---------- */
function attPlanMachines_(ymd, shift) {
  try {
    const rel = pmsRows_('pln').filter(r => r['Date'] instanceof Date && modFmt_(r['Date'], 'yyyy-MM-dd') === ymd && /^released/i.test(String(r['Status'])));
    const set = {}; rel.filter(r => String(r['Shift']) === shift).forEach(r => set[pmsNorm_(r['Machine'])] = 1);
    let names = {}; try { pmsMachines_().forEach(m => { names[pmsNorm_(m.code)] = pmsNorm_(m.name); names[pmsNorm_(m.name)] = pmsNorm_(m.code); }); } catch (e) {}
    Object.keys(set).forEach(k => { if (names[k]) set[names[k]] = 1; });
    return { has: rel.length > 0, set, n: Object.keys(set).length };
  } catch (e) { return { has: false, set: {}, n: 0, err: String(e.message || e) }; }
}
function attPlanRows_() {
  return attRows_('plan').map(r => ({ area: String(r['Area']), role: String(r['Role'] || ''), machine: String(r['Machine'] || ''), req: num_(r['Required']),
    link: String(r['Plan Link'] || 'Always'), note: String(r['Note'] || ''), row: r._row }));
}
function attRequired_(ymd, shift) {
  const P = attPlanMachines_(ymd, shift), tf = Object.keys(P.set).length > 0;
  const rows = attPlanRows_().map(p => {
    let need = p.req, why = '';
    if (P.has) {
      if (/^machine/i.test(p.link)) { need = P.set[pmsNorm_(p.machine)] ? p.req : 0; why = need ? 'planned' : 'not planned'; }
      else if (/any/i.test(p.link)) { need = tf ? p.req : 0; why = tf ? 'TF running' : 'no TF planned'; }
    }
    return Object.assign({}, p, { need, why });
  });
  return { rows, hasPlan: P.has, planned: Object.keys(P.set), err: P.err || '' };
}

/* =====================  Called from the web page  ===================== */
function attInit(token) {
  const { u, w } = attUser_(token, 'adv'), s = settingsMap_(), base = getSettings_(), book = attBook_(false);
  const L = attLabour_().filter(l => l.active).sort((a, b) => a.name.localeCompare(b.name));
  const out = { user: { name: u.name, username: u.username, rtype: u.rtype }, access: w, portalUrl: portalUrl_(), now: attNow_(), areas: ATT_AREAS, statuses: ATT_STATUS,
    company: { name: s['Company Name'] || base.name, short: base.short, logo: base.logo, tagline: s['Tagline'] || base.tagline || '' },
    sheetUrl: w.sir ? book.getUrl() : '',
    labour: w.att ? attLabour_() : L.map(l => ({ code: l.code, key: l.key, name: l.name, area: l.area })) };
  if (w.att) out.plan = attPlanRows_();
  return out;
}

/* shift board: per area required / present / extra / short (+ who is present) */
function attBoard(token, ymd, shift) {
  attUser_(token, 'att'); shift = attShift_(shift); attDate_(ymd);
  const R = attRequired_(ymd, shift), lab = {}; attLabour_().forEach(l => lab[l.key] = l);
  const day = attRows_('day').filter(r => attYmd_(r['Date']) === ymd && r['Shift'] === shift);
  const ext = attRows_('ext').filter(r => attYmd_(r['Date']) === ymd && r['Shift'] === shift);
  const A = {}; const area = a => A[a] = A[a] || { area: a, req: 0, present: 0, extra: 0, people: [], noOut: 0, lines: [] };
  R.rows.forEach(p => { const x = area(p.area); x.req += p.need; x.lines.push({ role: p.role, machine: p.machine, req: p.req, need: p.need, why: p.why }); });
  day.forEach(r => {
    const st = String(r['Status']); if (st !== 'P' && st !== 'HD') return;
    const l = lab[attCode_(r['Pay Code'])] || {}, a = String(r['Area'] || l.area || '') || 'Not set', x = area(a);
    x.present += st === 'HD' ? 0.5 : 1; if (r['Flag'] === 'No Out') x.noOut++;
    x.people.push({ code: String(r['Pay Code']), name: String(r['Name'] || l.name || ''), in: attTime_(r['In']), out: attTime_(r['Out']), st, flag: String(r['Flag'] || ''), src: String(r['Source'] || '') });
  });
  ext.forEach(r => { const x = area(String(r['Area'] || 'Other')); x.extra += num_(r['Count']); });
  const order = ATT_AREAS.concat(['Not set']);
  const areas = Object.keys(A).map(k => A[k]).map(x => Object.assign(x, { short: Math.max(0, x.req - x.present - x.extra), over: Math.max(0, x.present + x.extra - x.req) }))
    .sort((a, b) => (order.indexOf(a.area) + 1 || 99) - (order.indexOf(b.area) + 1 || 99));
  const T = areas.reduce((t, x) => ({ req: t.req + x.req, present: t.present + x.present, extra: t.extra + x.extra, short: t.short + x.short, noOut: t.noOut + x.noOut }), { req: 0, present: 0, extra: 0, short: 0, noOut: 0 });
  return { ymd, shift, areas, total: T, hasPlan: R.hasPlan, planned: R.planned, planErr: R.err,
    extras: ext.map(r => ({ id: r['ID'], area: r['Area'], count: num_(r['Count']), names: r['Names'], contractor: r['Contractor'], note: r['Note'], by: r['Entered By'] })) };
}

/* one shift: every active person + his row (if any) */
function attDay(token, ymd, shift) {
  attUser_(token, 'att'); shift = attShift_(shift); attDate_(ymd);
  const rows = {}; attRows_('day').filter(r => attYmd_(r['Date']) === ymd && r['Shift'] === shift).forEach(r => rows[attCode_(r['Pay Code'])] = {
    status: String(r['Status']), in: attTime_(r['In']), out: attTime_(r['Out']), hours: r['Hours'], area: String(r['Area'] || ''), flag: String(r['Flag'] || ''),
    src: String(r['Source'] || ''), note: String(r['Note'] || ''), by: String(r['Updated By'] || '') });
  const ext = attRows_('ext').filter(r => attYmd_(r['Date']) === ymd && r['Shift'] === shift)
    .map(r => ({ id: r['ID'], area: r['Area'], count: num_(r['Count']), names: r['Names'], contractor: r['Contractor'], note: r['Note'], by: r['Entered By'] }));
  return { ymd, shift, rows, extras: ext };
}

/* Shivam / Sirs: mark or correct one person (always Source = Manual → imports never touch it again) */
function attSave(token, d) {
  const { u } = attUser_(token, 'att'); rw_(u, true); d = d || {};
  const ymd = d.ymd, shift = attShift_(d.shift), date = attDate_(ymd), key = attCode_(d.code);
  if (!key) throw new Error('Pay Code missing');
  const st = String(d.status || ''); if (ATT_STATUS.indexOf(st) < 0) throw new Error('Pick a status');
  const l = attLabour_().find(x => x.key === key); if (!l) throw new Error('Pay Code ' + d.code + ' is not on the Labour list');
  const inT = attTime_(d.in), outT = attTime_(d.out);
  if (d.in && !inT) throw new Error('In time like 08:05'); if (d.out && !outT) throw new Error('Out time like 20:10');
  return attLock_(() => {
    const sh = attSheet_('day'), k = ymd + '|' + shift + '|' + key, ex = attRows_('day').find(r => r['Key'] === k);
    const o = { 'Key': k, 'Date': date, 'Shift': shift, 'Pay Code': l.code, 'Name': l.name, 'Area': d.area || l.area, 'Status': st, 'In': inT, 'Out': outT,
      'Hours': attHours_(inT, outT), 'Flag': (st === 'P' || st === 'HD') && inT && !outT ? 'No Out' : '', 'Source': 'Manual', 'Note': String(d.note || '').slice(0, 300),
      'Updated By': u.name, 'Updated At': new Date() };
    if (ex) sh.getRange(ex._row, 1, 1, attH_('day').length).setValues([attRowArr_('day', o)]); else modAppend_(sh, [attRowArr_('day', o)]);
    return { ok: true, flag: o['Flag'], hours: o['Hours'] };
  });
}
/* remove a person's row for that shift (e.g. marked by mistake) */
function attClear(token, ymd, shift, code) {
  const { u } = attUser_(token, 'att'); rw_(u, true); shift = attShift_(shift);
  return attLock_(() => { const k = ymd + '|' + shift + '|' + attCode_(code), ex = attRows_('day').find(r => r['Key'] === k);
    if (ex) attSheet_('day').deleteRow(ex._row); return { ok: true }; });
}

/* monthly grid: names × days 1–31 */
function attMonth(token, ym) {
  attUser_(token, 'att'); const m = String(ym || '').match(/^(\d{4})-(\d{2})$/); if (!m) throw new Error('Pick a month');
  const days = new Date(+m[1], +m[2], 0).getDate(), G = {};
  attRows_('day').filter(r => attYmd_(r['Date']).slice(0, 7) === ym).forEach(r => {
    const k = attCode_(r['Pay Code']), d = +attYmd_(r['Date']).slice(8, 10), g = G[k] = G[k] || {}, c = g[d] = g[d] || { s: [], f: 0 };
    const st = String(r['Status']); c.s.push(st === 'P' ? (r['Shift'] === 'Night' ? 'N' : 'D') : st); if (r['Flag'] === 'No Out') c.f = 1;
  });
  const people = attLabour_().filter(l => l.active || G[l.key]).sort((a, b) => (ATT_AREAS.indexOf(a.area) + 1 || 99) - (ATT_AREAS.indexOf(b.area) + 1 || 99) || a.name.localeCompare(b.name))
    .map(l => { const g = G[l.key] || {}, cells = []; let p = 0;
      for (let d = 1; d <= days; d++) { const c = g[d]; if (!c) { cells.push(null); continue; }
        const s = c.s.slice().sort(); p += s.reduce((t, x) => t + (x === 'D' || x === 'N' ? 1 : x === 'HD' ? 0.5 : 0), 0);
        cells.push({ t: s.join(''), f: c.f }); }
      return { code: l.code, name: l.name, area: l.area, cells, present: p }; });
  return { ym, days, people };
}

/* labour list: add / edit (Shivam + Sirs) */
function attLabourSave(token, d) {
  const { u } = attUser_(token, 'att'); rw_(u, true); d = d || {};
  const code = String(d.code || '').trim(), key = attCode_(code), name = String(d.name || '').trim();
  if (!key || !/^\d+$/.test(key)) throw new Error('Pay Code (numbers, as on the Time Notifier)'); if (!name) throw new Error('Name');
  return attLock_(() => {
    const sh = attSheet_('lab'), L = attLabour_(), ex = L.find(x => x.key === key);
    if (ex && d.isNew) throw new Error('Pay Code ' + code + ' is already used by ' + ex.name);
    const o = { 'Pay Code': ex ? ex.code : code, 'Name': name, 'Area': d.area || '', 'Role': d.role || '', 'Default Shift': d.shift || '', 'Phone': String(d.phone || '').replace(/[^\d+ ]/g, ''),
      'Active': d.active === false ? 'No' : 'Yes', 'Note': String(d.note || '').slice(0, 200) };
    if (ex) { const H = attH_('lab'); Object.keys(o).forEach(h => sh.getRange(ex.row, H.indexOf(h) + 1).setValue(o[h])); }
    else modAppend_(sh, [attRowArr_('lab', Object.assign(o, { 'Added By': u.name, 'Added At': new Date() }))]);
    return { ok: true, labour: attLabour_() };
  });
}

/* extra labour for a shift */
function attExtraSave(token, d) {
  const { u } = attUser_(token, 'att'); rw_(u, true); d = d || {};
  const date = attDate_(d.ymd), shift = attShift_(d.shift), n = Math.round(num_(d.count));
  if (!d.area) throw new Error('Pick the area'); if (n < 1 || n > 50) throw new Error('How many people (1–50)');
  return attLock_(() => { const sh = attSheet_('ext'); let mx = 0; attRows_('ext').forEach(r => { const m = String(r['ID']).match(/^EX-(\d+)$/); if (m) mx = Math.max(mx, +m[1]); });
    const id = 'EX-' + String(mx + 1).padStart(4, '0');
    modAppend_(sh, [attRowArr_('ext', { 'ID': id, 'Date': date, 'Shift': shift, 'Area': d.area, 'Count': n, 'Names': String(d.names || '').slice(0, 300), 'Contractor': String(d.contractor || '').slice(0, 100),
      'Note': String(d.note || '').slice(0, 200), 'Entered By': u.name, 'Entered At': new Date() })]);
    return { ok: true, id }; });
}
function attExtraDel(token, id) {

/* advances: Shivam, Seema Mam and the Sirs enter — only the Sirs see the list and balances */
function attAdvSave(token, d) {
  const { u, w } = attUser_(token, 'adv'); rw_(u, true); d = d || {};
  const date = attDate_(d.ymd), key = attCode_(d.code), amt = Math.round(num_(d.amount));
  const type = /^rec/i.test(String(d.type)) ? 'Recovery' : 'Advance';
  if (type === 'Recovery' && !w.sir) throw new Error('Only the Sirs enter a recovery');
  if (!(amt > 0)) throw new Error('Enter the amount');
  const l = attLabour_().find(x => x.key === key); if (!l) throw new Error('Pick the person');
  return attLock_(() => { const sh = attSheet_('adv'); let mx = 0; attRows_('adv').forEach(r => { const m = String(r['ID']).match(/^AD-(\d+)$/); if (m) mx = Math.max(mx, +m[1]); });
    const id = 'AD-' + String(mx + 1).padStart(4, '0');
    modAppend_(sh, [attRowArr_('adv', { 'ID': id, 'Date': date, 'Pay Code': l.code, 'Name': l.name, 'Type': type, 'Amount': amt, 'Note': String(d.note || '').slice(0, 200), 'Entered By': u.name, 'Entered At': new Date() })]);
    return { ok: true, id, name: l.name }; });   // amount is not sent back
}
function attAdvList(token) {
  attUser_(token, 'sir');
  const list = attRows_('adv').map(r => ({ id: r['ID'], ymd: attYmd_(r['Date']), code: String(r['Pay Code']), key: attCode_(r['Pay Code']), name: String(r['Name']), type: String(r['Type']),
    amt: num_(r['Amount']), note: String(r['Note'] || ''), by: String(r['Entered By'] || '') })).reverse();
  const B = {}; list.forEach(x => { const b = B[x.key] = B[x.key] || { code: x.code, name: x.name, given: 0, back: 0, last: '' };
    if (x.type === 'Recovery') b.back += x.amt; else b.given += x.amt; if (x.ymd > b.last) b.last = x.ymd; });
  const bal = Object.keys(B).map(k => Object.assign(B[k], { bal: B[k].given - B[k].back })).filter(b => b.bal !== 0 || b.given).sort((a, b) => b.bal - a.bal);
  return { list: list.slice(0, 300), bal };
}

/* manpower plan (Sirs) */
function attPlanSave(token, rows) {
  const { u } = attUser_(token, 'sir'); rw_(u, true);
  rows = (rows || []).filter(r => r && r.area);
  if (!rows.length) throw new Error('Nothing to save');
  return attLock_(() => { const sh = attSheet_('plan'), n = sh.getLastRow() - 3, H = attH_('plan');
    if (n > 0) sh.getRange(4, 1, n, H.length).clearContent();
    modAppend_(sh, rows.map(r => [r.area, r.role || '', r.machine || '', Math.max(0, Math.round(num_(r.req))), r.link || 'Always', r.note || '']));
    return { ok: true, plan: attPlanRows_() }; });
}

/* Time Notifier "Daily Attendance Report" — read in the browser, rows sent here.
 * p = { file, date:'yyyy-MM-dd', time:'HH:mm' (report time), rows:[{code,name,in,out}] } */
function attImport(token, p) {
  const { u } = attUser_(token, 'att'); rw_(u, true); p = p || {};
  const rdate = attDate_(p.date); if (rdate.getTime() > modDay_(new Date()).getTime() + 864e5) throw new Error('Report date is in the future');
  const rows = (p.rows || []).filter(r => attCode_(r.code)); if (!rows.length) throw new Error('No Pay Code rows found in the file');
  return attLock_(() => {
    const lab = attLabour_(), byKey = {}; lab.forEach(l => byKey[l.key] = l);
    const newL = [];
    rows.forEach(r => { const k = attCode_(r.code); if (!byKey[k] && /^\d+$/.test(k)) { const o = { code: String(r.code).trim(), key: k, name: String(r.name || '').trim(), area: '' }; byKey[k] = o; newL.push(o); } });
    if (newL.length) modAppend_(attSheet_('lab'), newL.map(o => attRowArr_('lab', { 'Pay Code': o.code, 'Name': o.name, 'Active': 'Yes', 'Note': 'new from import — set Area', 'Added By': u.name + ' (import)', 'Added At': new Date() })));
    const sh = attSheet_('day'), have = {}; attRows_('day').forEach(r => have[r['Key']] = r);
    const add = [], W = attH_('day').length; let present = 0, noOut = 0, kept = 0, upd = 0;
    rows.forEach(r => {
      const inT = attTime_(r.in), outT = attTime_(r.out); if (!inT) return;   // no punch → nothing written (absent = no row)
      const k0 = attCode_(r.code), l = byKey[k0], sf = attShiftOf_(p.date, inT), key = sf.ymd + '|' + sf.shift + '|' + k0, ex = have[key];
      if (ex && String(ex['Source']) === 'Manual') { kept++; return; }
      const o = { 'Key': key, 'Date': modYmd_(sf.ymd), 'Shift': sf.shift, 'Pay Code': l.code, 'Name': l.name, 'Area': ex ? String(ex['Area'] || l.area || '') : (l.area || ''),
        'Status': 'P', 'In': inT, 'Out': outT, 'Hours': attHours_(inT, outT), 'Flag': outT ? '' : 'No Out', 'Source': 'Import', 'Note': ex ? String(ex['Note'] || '') : '',
        'Updated By': u.name + ' (import)', 'Updated At': new Date() };
      present++; if (!outT) noOut++;
      if (ex) { sh.getRange(ex._row, 1, 1, W).setValues([attRowArr_('day', o)]); upd++; } else { add.push(attRowArr_('day', o)); have[key] = o; }
    });
    if (add.length) modAppend_(sh, add);
    modAppend_(attSheet_('log'), [attRowArr_('log', { 'At': new Date(), 'By': u.name, 'File': String(p.file || '').slice(0, 120), 'Report Date': rdate, 'Rows': rows.length, 'Present': present,
      'No Out': noOut, 'New Labour': newL.map(x => x.code + ' ' + x.name).join(', '), 'Kept Manual': kept, 'Note': (p.time ? 'report ' + p.time + ' · ' : '') + add.length + ' new rows, ' + upd + ' updated' })]);
    return { ok: true, rows: rows.length, present, noOut, kept, added: add.length, updated: upd, newLabour: newL.map(x => x.code + ' ' + x.name) };
  });
}
function attImportLog(token) {
  attUser_(token, 'att');
  return attRows_('log').slice(-15).reverse().map(r => ({ at: r['At'] instanceof Date ? modFmt_(r['At'], 'dd-MMM HH:mm') : '', by: r['By'], file: r['File'], date: r['Report Date'] instanceof Date ? modFmt_(r['Report Date'], 'dd-MMM-yyyy') : '',
    rows: r['Rows'], present: r['Present'], noOut: r['No Out'], newL: r['New Labour'], kept: r['Kept Manual'], note: r['Note'] }));
}
