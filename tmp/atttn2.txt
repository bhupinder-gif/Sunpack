/* ===================== TIME NOTIFIER — automatic pull (28-Oct-2026) =====================
 * Reads punches straight from timenotifier.com (no Excel). READ-ONLY: only Login, Get_Employee_List and Get_Punch_List are called.
 * Login is kept in Script Properties added by Bhupinder Sir: TN_COMPANY, TN_USER, TN_PASS (never in code / sheets).
 * Every 30 min (attTnJob, set by attTnSetup) → yesterday + today for every employee → pairs punches into shifts:
 *   first punch = In, last punch within 16 h = Out (punches < 10 min apart = one punch); shift from the In time (attShiftOf_).
 *   Night shift Out (next morning) is found because both days are read.
 * Rows go to the Daily tab with Source "Time Notifier"; Shivam's Manual rows are never touched.
 * Old Import / Time Notifier rows in the synced dates that no longer match a punch pair are removed (Manual rows never). */
const ATT_TN = 'https://timenotifier.com';
function attTnProps_() { return PropertiesService.getScriptProperties(); }
function attTnCreds_() {
  const P = attTnProps_(), c = { company: P.getProperty('TN_COMPANY'), user: P.getProperty('TN_USER'), pass: P.getProperty('TN_PASS') };
  if (!c.company || !c.user || !c.pass) throw new Error('Time Notifier login not added yet — Bhupinder Sir: Apps Script → Project Settings → Script properties → add TN_COMPANY, TN_USER, TN_PASS');
  return c;
}
function attTnCookies_(resp, jar) {
  const h = resp.getAllHeaders(); let sc = h['Set-Cookie'] || h['set-cookie'] || []; if (!Array.isArray(sc)) sc = [sc];
  sc.forEach(s => { const m = String(s).match(/^\s*([^=;]+)=([^;]*)/); if (m) jar[m[1]] = m[2]; });
  return jar;
}
function attTnJarStr_(jar) { return Object.keys(jar).map(k => k + '=' + jar[k]).join('; '); }
function attTnLogin_() {
  const c = attTnCreds_(), jar = {};
  attTnCookies_(UrlFetchApp.fetch(ATT_TN + '/Login/Index', { muteHttpExceptions: true, followRedirects: false }), jar);
  const r = UrlFetchApp.fetch(ATT_TN + '/Login/login', { method: 'post', payload: { company_code: c.company, Username: c.user, Pass: c.pass },
    headers: { Cookie: attTnJarStr_(jar) }, muteHttpExceptions: true, followRedirects: false });
  attTnCookies_(r, jar);
  const ck = attTnJarStr_(jar), list = attTnJson_(ck, '/Employee/select_EmployeeList');
  if (!Array.isArray(list)) throw new Error('Time Notifier login failed — check the Time Notifier login');
  const emps = list.filter(e => /active/i.test(String(e['STATUS'] || 'Active')) && e['PAY CODE'])
    .map(e => ({ id: String(e['Id']), code: String(e['PAY CODE']).trim(), card: String(e['CARD NO'] || '').trim(), name: String(e['NAME'] || '').trim(), night: /night/i.test(String(e['SHIFT NAME'] || '')) }));
  return { ck, emps };
}
function attTnJson_(ck, path, body) {
  const o = { method: body ? 'post' : 'get', headers: { Cookie: ck }, muteHttpExceptions: true, followRedirects: false };
  if (body) { o.contentType = 'application/json'; o.payload = JSON.stringify(body); }
  const t = UrlFetchApp.fetch(ATT_TN + path, o).getContentText();
  try { return JSON.parse(t); } catch (e) { return null; }
}
/* "27-10-2026 07:58:11" → { ymd, hm, t } */
function attTnStamp_(s) {
  const m = String(s || '').match(/^(\d{2})-(\d{2})-(\d{4})\s+(\d{2}):(\d{2})/); if (!m) return null;
  return { ymd: m[3] + '-' + m[2] + '-' + m[1], hm: m[4] + ':' + m[5], t: new Date(+m[3], +m[2] - 1, +m[1], +m[4], +m[5]).getTime() };
}
function attTnPairs_(P, night) {
  const seen = {}, q = [];
  P.filter(p => seen[p.t] ? false : (seen[p.t] = 1)).sort((a, b) => a.t - b.t).forEach(p => { if (!q.length || p.t - q[q.length - 1].t >= 10 * 60000) q.push(p); });   // < 10 min apart = one punch
  const segs = []; q.forEach((p, i) => { if (!i || p.t - q[i - 1].t > 20 * 36e5) segs.push([]); segs[segs.length - 1].push(p); });   // gap > 20 h (off day) → next punch is an In
  const out = [];
  segs.forEach((sg, si) => {
    let i = 0;
    if (si === 0) { const h = +sg[0].hm.slice(0, 2), isN = h >= 16 || h < 4; if (isN !== !!night) i = 1; }   // no gap seen yet: first punch must fit the person's Time Notifier shift, else it is the Out of an earlier shift
    while (i < sg.length) { const a = sg[i]; let j = i + 1, last = null; while (j < sg.length && sg[j].t - a.t <= 16 * 36e5) { last = sg[j]; j++; } out.push({ in: a, out: last }); i = j; }
  });
  return out;
}
/* pull punches for ymd from..to (max 10 days) and write the Daily tab */
function attTnSync_(from, to, by) {
  const d0 = modYmd_(from), d1 = modYmd_(to); if (!d0 || !d1 || d1 < d0) throw new Error('Pick the dates');
  const days = Math.round((d1 - d0) / 864e5) + 1; if (days > 7) throw new Error('Max 7 days per sync');
  const t0 = Date.now(), { ck, emps } = attTnLogin_(), E = emps, today = modDay_(new Date()).getTime();
  // read 7 days before (so the weekly-off gap shows who is on day / night) up to the day after (night-shift Out), never past today
  const dates = []; for (let t = d0.getTime() - 7 * 864e5; t <= Math.min(d1.getTime() + 864e5, today); t += 864e5) dates.push(modFmt_(new Date(t), 'yyyy-MM-dd'));
  const C = CacheService.getScriptCache(), old = dates.filter(y => modYmd_(y).getTime() < today - 864e5);   // days before yesterday are kept 6 h in the cache
  const reqs = [], who = [], P = {};
  E.forEach(e => dates.forEach(ymd => {
    const ckey = 'tnp_' + e.id + '_' + ymd;
    if (old.indexOf(ymd) > -1) { const hit = C.get(ckey); if (hit) { try { (P[e.code] = P[e.code] || []).push(...JSON.parse(hit)); return; } catch (x) {} } }
    reqs.push({ url: ATT_TN + '/MissPunchEntry/Get_Punch_List', method: 'post', contentType: 'application/json', muteHttpExceptions: true, followRedirects: false,
      headers: { Cookie: ck }, payload: JSON.stringify({ ID: '0', EMP_ID: e.id, PAYCODE: e.code, CARDNO: e.card || attCode_(e.code), OFFICEPUNCH: ymd + ' 00:00' }) });
    who.push({ e, ymd, ckey, keep: old.indexOf(ymd) > -1 });
  }));
  let todo = reqs.map((r, i) => i);
  for (let round = 0; round < 3 && todo.length; round++) {   // batches of 10 (the site refuses many at once) · failed ones retried twice
    if (round) Utilities.sleep(2000); const fail = [];
    for (let i = 0; i < todo.length; i += 10) {
      const ids = todo.slice(i, i + 10);
      UrlFetchApp.fetchAll(ids.map(x => reqs[x])).forEach((r, k) => { let j = null; try { j = JSON.parse(r.getContentText()); } catch (x) {}
        if (!Array.isArray(j)) { fail.push(ids[k]); return; } const w = who[ids[k]], L = j.map(x => attTnStamp_(x['OFFICE PUNCH'])).filter(Boolean);
        (P[w.e.code] = P[w.e.code] || []).push(...L); if (w.keep) { try { C.put(w.ckey, JSON.stringify(L), 21600); } catch (x) {} } });
      if (Date.now() - t0 > 300000) throw new Error('Time Notifier is slow today — stopped after 5 minutes, nothing changed. Try fewer days.');
    }
    todo = fail;
  }
  if (todo.length) throw new Error('Time Notifier did not answer for ' + todo.length + ' of ' + reqs.length + ' look-ups — nothing changed, try again later');
  const fromS = modFmt_(d0, 'yyyy-MM-dd'), toS = modFmt_(d1, 'yyyy-MM-dd'), now = Date.now();
  return attLock_(() => {
    const lab = attLabour_(), byKey = {}; lab.forEach(l => byKey[l.key] = l); const newL = [];
    E.forEach(e => { const k = attCode_(e.code); if (!byKey[k] && /^\d+$/.test(k) && (P[e.code] || []).length) { const o = { code: e.code, key: k, name: e.name, area: '' }; byKey[k] = o; newL.push(o); } });
    if (newL.length) modAppend_(attSheet_('lab'), newL.map(o => attRowArr_('lab', { 'Pay Code': o.code, 'Name': o.name, 'Active': 'Yes', 'Note': 'new from Time Notifier — set Area', 'Added By': 'Time Notifier', 'Added At': new Date() })));
    const sh = attSheet_('day'), W = attH_('day').length, rows = attRows_('day'), have = {}; rows.forEach(r => have[r['Key']] = r);
    const want = {}, add = []; let upd = 0, kept = 0, same = 0, noOut = 0;
    const nightOf = {}; E.forEach(e => nightOf[e.code] = e.night);
    Object.keys(P).forEach(code => {
      const k0 = attCode_(code), l = byKey[k0]; if (!l) return;
      attTnPairs_(P[code], nightOf[code]).forEach(pr => {
        const sf = attShiftOf_(pr.in.ymd, pr.in.hm); if (sf.ymd < fromS || sf.ymd > toS) return;
        const key = sf.ymd + '|' + sf.shift + '|' + k0; want[key] = 1; const ex = have[key];
        if (ex && String(ex['Source']) === 'Manual') { kept++; return; }
        const end = modYmd_(sf.ymd).getTime() + (sf.shift === 'Day' ? 20 : 32) * 36e5, outT = pr.out ? pr.out.hm : '';
        const flag = !outT && now > end + 2 * 36e5 ? 'No Out' : '';
        if (flag) noOut++;
        if (ex && attTime_(ex['In']) === pr.in.hm && attTime_(ex['Out']) === outT && String(ex['Flag'] || '') === flag && String(ex['Source']) === 'Time Notifier') { same++; return; }
        const o = { 'Key': key, 'Date': modYmd_(sf.ymd), 'Shift': sf.shift, 'Pay Code': l.code, 'Name': l.name, 'Area': ex ? String(ex['Area'] || '') : '',
          'Status': 'P', 'In': pr.in.hm, 'Out': outT, 'Hours': attHours_(pr.in.hm, outT), 'Flag': flag, 'Source': 'Time Notifier', 'Note': ex ? String(ex['Note'] || '') : '',
          'Updated By': by || 'Auto (Time Notifier)', 'Updated At': new Date() };
        if (ex) { sh.getRange(ex._row, 1, 1, W).setValues([attRowArr_('day', o)]); upd++; } else { add.push(attRowArr_('day', o)); have[key] = o; }
      });
    });
    // old Import / Time Notifier rows in these dates without a matching punch pair → removed (Manual never)
    const synced = {}; E.forEach(e => synced[attCode_(e.code)] = 1);
    const stale = rows.filter(r => { const y = attYmd_(r['Date']); return y >= fromS && y <= toS && /^(Import|Time Notifier)$/.test(String(r['Source'])) && synced[attCode_(r['Pay Code'])] && !want[r['Key']]; })
      .map(r => r._row).sort((a, b) => b - a);
    stale.forEach(rw => sh.deleteRow(rw));
    if (add.length) modAppend_(sh, add);
    const res = { at: modFmt_(new Date(), 'dd-MMM HH:mm'), from: fromS, to: toS, people: E.length, added: add.length, updated: upd, same, kept, removed: stale.length, noOut,
      newLabour: newL.map(x => x.code + ' ' + x.name), secs: Math.round((Date.now() - t0) / 1000), by: by || 'auto' };
    attTnProps_().setProperty('ATT_TN_LAST', JSON.stringify(res));
    if (add.length || upd || stale.length) dataChanged_();
    return res;
  });
}
/* every 30 min (trigger) — yesterday + today */
function attTnJob() {
  const now = new Date(), y = modFmt_(new Date(now.getTime() - 864e5), 'yyyy-MM-dd'), t = modFmt_(now, 'yyyy-MM-dd');
  try { const r = attTnSync_(y, t, 'Auto (Time Notifier)'); console.log('TN sync ' + JSON.stringify(r)); }
  catch (e) { attTnProps_().setProperty('ATT_TN_LAST', JSON.stringify({ at: modFmt_(new Date(), 'dd-MMM HH:mm'), error: String(e.message || e) })); console.log('TN sync failed: ' + e); }
}
/* Run ONCE from the editor after the 3 Script Properties are added */
function attTnSetup() {
  ScriptApp.getProjectTriggers().filter(x => x.getHandlerFunction() === 'attTnJob').forEach(x => ScriptApp.deleteTrigger(x));
  ScriptApp.newTrigger('attTnJob').timeBased().everyMinutes(30).create();
  attTnJob();
  return attTnProps_().getProperty('ATT_TN_LAST');
}
/* page: status + Sync now (Shivam / Sirs) */
function attTnStatus(token) {
  attUser_(token, 'att'); let last = null; try { last = JSON.parse(attTnProps_().getProperty('ATT_TN_LAST') || 'null'); } catch (e) {}
  const P = attTnProps_(), ready = !!(P.getProperty('TN_COMPANY') && P.getProperty('TN_USER') && P.getProperty('TN_PASS'));
  return { ready, auto: ScriptApp.getProjectTriggers().some(x => x.getHandlerFunction() === 'attTnJob'), last };
}
function attTnSyncNow(token, from, to) {
  const { u } = attUser_(token, 'att'); rw_(u, true);
  return attTnSync_(from, to, u.name + ' (Time Notifier)');
}
